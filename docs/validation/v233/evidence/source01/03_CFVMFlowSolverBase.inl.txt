  1149 template <class V, ENUM_REGIME FlowRegime>
  1150 void CFVMFlowSolverBase<V, FlowRegime>::BC_Sym_Plane(CGeometry* geometry, CSolver** solver_container, CNumerics* conv_numerics,
  1151                                                      CNumerics* visc_numerics, CConfig* config, unsigned short val_marker) {
  1152   SU2_ZONE_SCOPED
  1153 
  1154   const bool implicit = (config->GetKind_TimeIntScheme() == EULER_IMPLICIT);
  1155   const bool ideal_gas = (config->GetKind_FluidModel() == STANDARD_AIR) ||
  1156                          (config->GetKind_FluidModel() == IDEAL_GAS);
  1157   const auto iVel = prim_idx.Velocity();
  1158 
  1159   /*--- Blazek chapter 8.:
  1160    * The components of the momentum residual normal to the symmetry plane are zeroed out.
  1161    * The gradients have already been corrected acording to Eq. (8.40).
  1162    * Contrary to Blazek we keep some scalar fluxes computed on the boundary to improve stability (see below). ---*/
  1163 
  1164   /*--- Loop over all the vertices on this boundary marker. ---*/
  1165 
  1166   SU2_OMP_FOR_DYN(OMP_MIN_SIZE)
  1167   for (auto iVertex = 0ul; iVertex < geometry->nVertex[val_marker]; iVertex++) {
  1168     const auto iPoint = geometry->vertex[val_marker][iVertex]->GetNode();
  1169 
  1170     /*--- Halo points do not need to be considered. ---*/
  1171     if (!geometry->nodes->GetDomain(iPoint)) continue;
  1172 
  1173     /*--- Get the normal of the current symmetry. This may be the original normal of the vertex
  1174      * or a modified normal if there are intersecting symmetries. ---*/
  1175 
  1176     su2double Normal[MAXNDIM] = {}, UnitNormal[MAXNDIM] = {};
  1177     geometry->vertex[val_marker][iVertex]->GetNormal(Normal);
  1178     const auto it = geometry->symmetryNormals[val_marker].find(iVertex);
  1179 
  1180     if (it != geometry->symmetryNormals[val_marker].end()) {
  1181       for (auto iDim = 0u; iDim < nDim; iDim++) UnitNormal[iDim] = it->second[iDim];
  1182     } else {
  1183       const su2double Area = GeometryToolbox::Norm(nDim, Normal);
  1184       for (auto iDim = 0u; iDim < nDim; iDim++) UnitNormal[iDim] = Normal[iDim] / Area;
  1185     }
  1186 
  1187     /*--- Energy terms due to grid movement (aka work of pressure forces). ---*/
  1188     if (dynamic_grid) {
  1189       su2double* V_reflected = GetCharacPrimVar(val_marker, iVertex);
  1190 
  1191       conv_numerics->SetGridVel(geometry->nodes->GetGridVel(iPoint),
  1192                                 geometry->nodes->GetGridVel(iPoint));
  1193 
  1194       /*--- Normal vector for this vertex (negate for outward convention). ---*/
  1195       for (auto iDim = 0u; iDim < nDim; iDim++) Normal[iDim] = -Normal[iDim];
  1196       conv_numerics->SetNormal(Normal);
  1197 
  1198       for (auto iVar = 0u; iVar < nPrimVar; iVar++)
  1199         V_reflected[iVar] = nodes->GetPrimitive(iPoint, iVar);
  1200 
  1201       su2double ProjVelocity_i = nodes->GetProjVel(iPoint, UnitNormal);
  1202       /*--- Adjustment to v.n due to grid movement. ---*/
  1203       ProjVelocity_i -= GeometryToolbox::DotProduct(nDim, geometry->nodes->GetGridVel(iPoint), UnitNormal);
  1204 
  1205       for (auto iDim = 0u; iDim < nDim; iDim++)
  1206         V_reflected[iDim + iVel] = nodes->GetVelocity(iPoint, iDim) - ProjVelocity_i * UnitNormal[iDim];
  1207 
  1208       /*--- Get current solution at this boundary node. ---*/
  1209       const su2double* V_domain = nodes->GetPrimitive(iPoint);
  1210 
  1211       /*--- Set Primitive and Secondary for numerics class. ---*/
  1212       conv_numerics->SetPrimitive(V_domain, V_reflected);
  1213       if (!ideal_gas) {
  1214         conv_numerics->SetSecondary(nodes->GetSecondary(iPoint), nodes->GetSecondary(iPoint));
  1215       }
  1216 
  1217       /*--- Compute the residual using an upwind scheme. ---*/
  1218       auto residual = conv_numerics->ComputeResidual(config);
  1219 
  1220       /*--- Use just the energy fluxes to update the residual, adding the others would
  1221        * increase numerical diffusion which we wish to avoid if possible. ---*/
  1222       for (auto iVar = iVel + nDim; iVar < nVar; iVar++) {
  1223         LinSysRes(iPoint, iVar) += residual.residual[iVar];
  1224       }
  1225       if (implicit) {
  1226         auto* block = Jacobian.GetBlock(iPoint, iPoint);
  1227         /*--- But in the Jacobian we also include the mass flux, this allows some cases with
  1228          * motion to use larger CFL, for example pywrapper_translating_naca0012. ---*/
  1229         for (auto iVar = 0u; iVar < nVar; iVar++) {
  1230           if (iVar < iVel || iVar >= iVel + nDim) {
  1231             for (auto jVar = 0u; jVar < nVar; jVar++) {
  1232               block[iVar * nVar + jVar] += SU2_TYPE::GetValue(residual.jacobian_i[iVar][jVar]);
  1233             }
  1234           }
  1235         }
  1236       }
  1237     }
  1238 
  1239     /*--- Explicitly set the velocity components normal to the symmetry plane to zero.
  1240      * This is necessary because the modification of the residual leaves the problem
  1241      * underconstrained (the normal residual is zero regardless of the normal velocity). ---*/
  1242 
  1243     su2double* solutionOld = nodes->GetSolution_Old(iPoint);
  1244 
  1245     su2double gridVel[MAXNDIM] = {};
  1246     if (dynamic_grid) {
  1247       for (auto iDim = 0u; iDim < nDim; iDim++) {
  1248         gridVel[iDim] = geometry->nodes->GetGridVel(iPoint)[iDim];
  1249       }
  1250       if (FlowRegime == ENUM_REGIME::COMPRESSIBLE) {
  1251         for(auto iDim = 0u; iDim < nDim; iDim++) {
  1252           /*--- Multiply by density since we are correcting conservative variables. ---*/
  1253           gridVel[iDim] *= nodes->GetDensity(iPoint);
  1254         }
  1255       }
  1256     }
  1257     su2double vp = 0.0;
  1258     for (auto iDim = 0u; iDim < nDim; iDim++) {
  1259       vp += (solutionOld[iVel + iDim] - gridVel[iDim]) * UnitNormal[iDim];
  1260     }
  1261     for (auto iDim = 0u; iDim < nDim; iDim++) {
  1262       solutionOld[iVel + iDim] -= vp * UnitNormal[iDim];
  1263     }
  1264 
  1265     /*--- Keep only the tangential part of the momentum residuals. ---*/
  1266     su2double normalRes = 0.0;
  1267     for (auto iDim = 0u; iDim < nDim; iDim++) {
  1268       normalRes += LinSysRes(iPoint, iVel + iDim) * UnitNormal[iDim];
  1269     }
  1270     for (auto iDim = 0u; iDim < nDim; iDim++) {
  1271       LinSysRes(iPoint, iVel + iDim) -= normalRes * UnitNormal[iDim];
  1272     }
  1273 
  1274     /*--- Jacobian contribution for implicit integration. ---*/
  1275     if (implicit) {
  1276       /*--- Modify the Jacobians according to the modification of the residual
  1277        * J_new = (I - n * n^T) * J where n = {0, nx, ny, nz, 0, ...} ---*/
  1278       su2double mat[MAXNVAR * MAXNVAR] = {};
  1279 
  1280       for (auto iVar = 0u; iVar < nVar; iVar++)
  1281         mat[iVar * nVar + iVar] = 1;
  1282       for (auto iDim = 0u; iDim < nDim; iDim++)
  1283         for (auto jDim = 0u; jDim < nDim; jDim++)
  1284           mat[(iDim + iVel) * nVar + jDim + iVel] -= UnitNormal[iDim] * UnitNormal[jDim];
  1285 
  1286       auto ModifyJacobian = [&](const unsigned long jPoint) {
  1287         su2double jac[MAXNVAR * MAXNVAR], newJac[MAXNVAR * MAXNVAR];
  1288         const auto view = Jacobian.GetBlockView(iPoint, jPoint);
  1289         if (!view) return;
  1290         for (auto iVar = 0u; iVar < nVar; iVar++)
  1291           for (auto jVar = 0u; jVar < nVar; jVar++) jac[iVar * nVar + jVar] = view(iVar, jVar);
  1292 
  1293         CBlasStructure().gemm(nVar, nVar, nVar, mat, jac, newJac, config);
  1294 
  1295         Jacobian.SetBlock(iPoint, jPoint, newJac);
  1296       };
  1297       ModifyJacobian(iPoint);
  1298       for (size_t iNeigh = 0; iNeigh < geometry->nodes->GetnPoint(iPoint); ++iNeigh) {
  1299         ModifyJacobian(geometry->nodes->GetPoint(iPoint, iNeigh));
  1300       }
  1301     }
  1302 
  1303     /*--- Correction for multigrid. ---*/
  1304     normalRes = 0.0;
  1305     su2double* Res_TruncError = nodes->GetResTruncError(iPoint);
  1306     for (auto iDim = 0u; iDim < nDim; iDim++) {
  1307       normalRes += Res_TruncError[iVel + iDim] * UnitNormal[iDim];
  1308     }
  1309     for (auto iDim = 0u; iDim < nDim; iDim++) {
  1310       Res_TruncError[iVel + iDim] -= normalRes * UnitNormal[iDim];
  1311     }
  1312   }
  1313   END_SU2_OMP_FOR
  1314 
  1315 }

  1317 template <class V, ENUM_REGIME R>
  1318 void CFVMFlowSolverBase<V, R>::BC_Periodic(CGeometry* geometry, CSolver** solver_container, CNumerics* numerics,
  1319                                            CConfig* config) {
  1320   SU2_ZONE_SCOPED
  1321 
  1322   /*--- Complete residuals for periodic boundary conditions. We loop over
  1323    the periodic BCs in matching pairs so that, in the event that there are
  1324    adjacent periodic markers, the repeated points will have their residuals
  1325    accumulated correctly during the communications. For implicit calculations,
  1326    the Jacobians and linear system are also correctly adjusted here. ---*/
  1327 
  1328   for (unsigned short iPeriodic = 1; iPeriodic <= config->GetnMarker_Periodic() / 2; iPeriodic++) {
  1329     InitiatePeriodicComms(geometry, config, iPeriodic, PERIODIC_RESIDUAL);
  1330     CompletePeriodicComms(geometry, config, iPeriodic, PERIODIC_RESIDUAL);
  1331   }
  1332 }

  1339 
  1340   const bool ideal_gas = config->GetKind_FluidModel() == STANDARD_AIR || config->GetKind_FluidModel() == IDEAL_GAS;
  1341

  1347
