  2512         else if (fluid_donor && fluid_target) {
  2513           /*--- Interface handling for turbomachinery applications. ---*/
  2514           if (config[donor]->GetBoolTurbomachinery()) {
  2515             auto interfaceIndex = donor+target; // Here we assume that the interfaces at each side are the same kind
  2516             switch (config[donor]->GetKind_TurboInterface(interfaceIndex)) {
  2517               case TURBO_INTERFACE_KIND::MIXING_PLANE: {
  2518                 interpolation[donor][target] = unique_ptr<CInterpolator>(CInterpolatorFactory::CreateInterpolator(
  2519                                        geometry, config, interpolation[target][donor].get(), donor, target, true));
  2520                 string fname = "TURBOMACHINERY/Mixing_Plane_Interpolator_Donor_" + to_string(donor) + "_Target_" + to_string(target) + ".dat";
  2521                 interpolation[donor][target]->WriteInterpolationDetails(fname, config);
  2522                 if (rank == MASTER_NODE) cout << " Transferring ";
  2523                 interface_type = MIXING_PLANE;
  2524                 auto nVar = solver[donor][INST_0][MESH_0][FLOW_SOL]->GetnVar();
  2525                 interface[donor][target] = new CMixingPlaneInterface(nVar, 0);
  2526                 if (rank == MASTER_NODE) cout << " Using a mixing-plane interface from donor zone " << donor << " to target zone " << target << "." << endl;
  2527                 break;
  2528               }
  2529               case TURBO_INTERFACE_KIND::FROZEN_ROTOR: {
  2530                 interpolation[donor][target] = unique_ptr<CInterpolator>(CInterpolatorFactory::CreateInterpolator(
  2531                                        geometry, config, interpolation[target][donor].get(), donor, target, false));
  2532                 if (rank == MASTER_NODE) cout << " Transferring ";
  2533                 interface_type = SLIDING_INTERFACE;
  2534                 auto nVar = solver[donor][INST_0][MESH_0][FLOW_SOL]->GetnPrimVar();
  2535                 interface[donor][target] = new CSlidingInterface(nVar, 0);
  2536                 if (rank == MASTER_NODE) cout << " Using a fluid interface interface from donor zone " << donor << " to target zone " << target << "." << endl;
  2537               }
  2538             }
  2539           }
  2540           else{
  2541             interface_type = SLIDING_INTERFACE;
  2542             auto nVar = solver[donor][INST_0][MESH_0][FLOW_SOL]->GetnPrimVar();
  2543               interface[donor][target] = new CSlidingInterface(nVar, 0);
  2544               if (rank == MASTER_NODE) cout << " Sliding interface." << endl;
  2545           }
  2546         }
  2547         else if (heat_donor || heat_target) {
  2548           interface_type = GetChtInterfaceType(heat_donor, heat_target);
  2549           interface[donor][target] = MakeChtInterface(interface_type);
  2550         }
  2551         else {
  2552           if (solver[donor][INST_0][MESH_0][FLOW_SOL] == nullptr)
  2553             SU2_MPI::Error("Could not determine the number of variables for transfer.", CURRENT_FUNCTION);
  2554 
  2555           interface_type = CONSERVATIVE_VARIABLES;
  2556           auto nVar = solver[donor][INST_0][MESH_0][FLOW_SOL]->GetnVar();
  2557           interface[donor][target] = new CConservativeVarsInterface(nVar, 0);
  2558           if (rank == MASTER_NODE) cout << " Generic conservative variables." << endl;
  2559         }
  2560       }
  2561 
  2562     }
