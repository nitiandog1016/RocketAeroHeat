   289 void CMultizoneDriver::RunGaussSeidel() {
   290   SU2_ZONE_SCOPED
   291 
   292   unsigned short UpdateMesh;
   293   bool DeformMesh = false;
   294 
   295   for (iZone = 0; iZone < nZone; iZone++) {
   296     config_container[iZone]->SetOuterIter(0ul);
   297   }
   298 
   299   /*--- Loop over the number of outer iterations ---*/
   300   for (auto iOuter_Iter = 0ul; iOuter_Iter < driver_config->GetnOuter_Iter(); iOuter_Iter++) {
   301 
   302     /*--- Loop over the number of zones (IZONE) ---*/
   303     for (iZone = 0; iZone < nZone; iZone++) {
   304 
   305       /*--- In principle, the mesh does not need to be updated ---*/
   306       UpdateMesh = 0;
   307 
   308       /*--- Set the OuterIter ---*/
   309       config_container[iZone]->SetOuterIter(iOuter_Iter);
   310       config_container[iZone]->Set_StartTime(SU2_MPI::Wtime());
   311       driver_config->SetOuterIter(iOuter_Iter);
   312 
   313       /*--- Transfer from all the remaining zones ---*/
   314       for (auto jZone = 0u; jZone < nZone; jZone++){
   315         /*--- The target zone is iZone ---*/
   316         if (jZone != iZone){
   317           DeformMesh = TransferData(jZone, iZone);
   318           if (DeformMesh) UpdateMesh+=1;
   319         }
   320       }
   321       /*--- If a mesh update is required due to the transfer of data ---*/
   322       if (UpdateMesh > 0) DynamicMeshUpdate(iZone, TimeIter);
   323 
   324       /*--- Iterate the zone as a block, either to convergence or to a max number of iterations ---*/
   325       iteration_container[iZone][INST_0]->Solve(output_container[iZone], integration_container, geometry_container,
   326                                                 solver_container, numerics_container, config_container,
   327                                                 surface_movement, grid_movement, FFDBox, iZone, INST_0);
   328 
   329       /*--- A corrector step can help preventing numerical instabilities ---*/
   330       Corrector(iZone);
   331 
   332     }
   333 
   334     if (OuterConvergence(iOuter_Iter)) break;
   335 
   336   }
   337 
   338 }
