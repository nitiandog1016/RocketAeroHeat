3183: /**
3184:  * Compute perturbation size estimate for real-valued Frechet derivative.
3185:  *
3186:  * REFERENCE: Knoll, D.A. and McHugh, P.R.,
3187:  *            Newton-Krylov Methods Applied to a System of Convection-Diffusion-Reaction Equations,
3188:  *            Computer Physics Communications,
3189:  *            1994
3190:  *
3191:  * Authors: KAD and RJG
3192:  * Date: 2024-02-29
3193:  */
3194: double computePerturbationSize()
3195: {
3196:     // perturbation constant, Knoll and McHugh suggest a value on the order of the square
3197:     // root of machine roundoff, however, we have observed that this value can be somewhat
3198:     // problem dependent.
3199:     auto a = nkCfg.frechetDerivativePerturbation;
3200: 
3201:     // calculate L2 norm of zed vector
3202:     number z_L2 = 0.0;
3203:     mixin(dotOverBlocks("z_L2", "zed", "zed"));
3204:     version(mpi_parallel) {
3205:         MPI_Allreduce(MPI_IN_PLACE, &(z_L2.re), 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
3206:     }
3207:     z_L2 = sqrt(z_L2);
3208:     // When the L2 norm of the zed vector is small, the sigma calculated from this routine
3209:     // has been observed to result in noisy convergence as observed in the global residual.
3210:     // We enforce a minimum value to guard against this behaviour.
3211:     // From some experimentation a minimum value of sqrt(1.0e-07) appears sufficient.
3212:     double min_z_L2 = sqrt(1.0e-07);
3213:     if (z_L2.re <= min_z_L2) { z_L2.re = min_z_L2; }
3214: 
3215:     // compute sum of individual component perturbations, eps_i,
3216:     // where eps_i is calculated using Eq. 12, i.e. eps_i = a*|var_i| + a
3217:     double eps_sum = 0.0;
3218:     double nvars = 0.0;
3219:     foreach (blk; localFluidBlocks) {
3220:         foreach (cell; blk.cells) {
3221:             foreach (var; cell.U[0]) {
3222:                 eps_sum += a*fabs(var.re) + a;
3223:                 nvars += 1;
3224:             }
3225:         }
3226:     }
3227:     version(mpi_parallel) {
3228:         MPI_Allreduce(MPI_IN_PLACE, &(eps_sum.re), 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
3229:     }
3230:     version(mpi_parallel) {
3231:         MPI_Allreduce(MPI_IN_PLACE, &(nvars.re), 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
3232:     }
3233: 
3234:     // compute perturbation parameter via Eq. 11
3235:     double sigma = eps_sum/(nvars*z_L2.re);
3236: 
3237:     return sigma;
3238: }
