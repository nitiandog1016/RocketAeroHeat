2567: void CEulerSolver::ComputeUnderRelaxationFactor(const CConfig* config) {
2568:   SU2_ZONE_SCOPED
2569: 
2570:   /*--- Loop over the solution update given by relaxing the linear system for this
2571:    * nonlinear iteration and impose a limit on the maximum percentage that the
2572:    * density and static energy can change. */
2573: 
2574:   const su2double allowableRatio = config->GetMaxUpdateFractionFlow();
2575: 
2576:   SU2_OMP_FOR_STAT(omp_chunk_size)
2577:   for (unsigned long iPoint = 0; iPoint < nPointDomain; iPoint++) {
2578:     su2double ratio = fabs(LinSysSol(iPoint, 0)) / max(nodes->GetSolution(iPoint, 0), EPS);
2579:     su2double e_old = nodes->GetSolution(iPoint, nVar - 1);
2580:     su2double e_new = e_old + LinSysSol(iPoint, nVar - 1);
2581:     /*--- rho*e = rho*E - |rho*u|^2 / (2*rho), with the old and the proposed new density. ---*/
2582:     /*--- Materialize CoDi expressions: GetSolution returns temporary active scalars. ---*/
2583:     const su2double rho_old = max(nodes->GetSolution(iPoint, 0), EPS);
2584:     const su2double rho_new = max(rho_old + LinSysSol(iPoint, 0), EPS);
2585:     for (auto jVar = 1u; jVar <= nDim; jVar++) {
2586:       e_old -= 0.5 * pow(nodes->GetSolution(iPoint, jVar), 2) / rho_old;
2587:       e_new -= 0.5 * pow(nodes->GetSolution(iPoint, jVar) + LinSysSol(iPoint, jVar), 2) / rho_new;
2588:     }
2589:     ratio = fmax(ratio, fabs(e_new - e_old) / max(e_old, EPS));
2590: 
2591:     su2double localUnderRelaxation = fmin(allowableRatio / fmax(ratio, EPS), 1);
2592: 
2593:     /* Threshold the relaxation factor in the event that there is
2594:      a very small value. This helps avoid catastrophic crashes due
2595:      to non-realizable states by canceling the update. */
2596: 
2597:     if (localUnderRelaxation < 1e-10) localUnderRelaxation = 0.0;
2598: 
2599:     nodes->SetUnderRelaxation(iPoint, localUnderRelaxation);
