5400           TangVelocity= -sqrt(Velocity2_e)*Flow_Dir[1];

5402           Velocity_e[1]= UnitNormal[1]*NormalVelocity + UnitNormal[0]*TangVelocity;

5410         StaticEnergy_e = GetFluidModel()->GetStaticEnergy();
5411         Energy_e = StaticEnergy_e + 0.5 * Velocity2_e;

5415       case STATIC_SUPERSONIC_INFLOW_PT:
5416 
5417         /*--- Retrieve the specified total conditions for this boundary. ---*/
5418         if (gravity) P_static = config->GetRiemann_Var1(Marker_Tag) - geometry->nodes->GetCoord(iPoint, nDim-1)*STANDARD_GRAVITY;/// check in which case is true (only freesurface?)
5419         else P_static  = config->GetRiemann_Var1(Marker_Tag);
5420         T_static  = config->GetRiemann_Var2(Marker_Tag);
5421         Mach = config->GetRiemann_FlowDir(Marker_Tag);
5422 
5423         /*--- Non-dim. the inputs if necessary. ---*/
5424         P_static /= config->GetPressure_Ref();
5425         T_static /= config->GetTemperature_Ref();
5426 
5427         /* --- Computes the total state --- */
5428         GetFluidModel()->SetTDState_PT(P_static, T_static);
5429 
5430         /* --- Compute the boundary state u_e --- */
5431         for (auto iDim = 0u; iDim < nDim; iDim++)

5433 
5434         Velocity2_e = GeometryToolbox::SquaredNorm(nDim, Velocity_e);
5435         Density_e = GetFluidModel()->GetDensity();
5436         StaticEnergy_e = GetFluidModel()->GetStaticEnergy();
5437         Energy_e = StaticEnergy_e + 0.5 * Velocity2_e;
5438         if (tkeNeeded) Energy_e += GetTke_Inf();
5439         break;
5440 
5441       case STATIC_SUPERSONIC_INFLOW_PD:
5442 
5443         /*--- Retrieve the specified total conditions for this boundary. ---*/
5444 
5445         if (gravity) P_static = config->GetRiemann_Var1(Marker_Tag) - geometry->nodes->GetCoord(iPoint, nDim-1)*STANDARD_GRAVITY;/// check in which case is true (only freesurface?)
5446         else P_static  = config->GetRiemann_Var1(Marker_Tag);
5447         Density_static  = config->GetRiemann_Var2(Marker_Tag);
5448         Mach = config->GetRiemann_FlowDir(Marker_Tag);
5449 
5450         /*--- Non-dim. the inputs if necessary. ---*/
5451         P_static /= config->GetPressure_Ref();
5452         Density_static /= config->GetDensity_Ref();
5453 
5454         /* --- Computes the total state --- */
5455         GetFluidModel()->SetTDState_Prho(P_static, Density_static);
5456 
5457         /* --- Compute the boundary state u_e --- */
5458         for (auto iDim = 0u; iDim < nDim; iDim++)
5459           Velocity_e[iDim] = Mach[iDim]*GetFluidModel()->GetSoundSpeed();
5460 
5461         Velocity2_e = GeometryToolbox::SquaredNorm(nDim, Velocity_e);
5462         Density_e = GetFluidModel()->GetDensity();
5463         StaticEnergy_e = GetFluidModel()->GetStaticEnergy();

5465         if (tkeNeeded) Energy_e += GetTke_Inf();
5466         break;
5467 
5468       case DENSITY_VELOCITY:

5470         /*--- Retrieve the specified density and velocity magnitude ---*/
5471         Density_e  = config->GetRiemann_Var1(Marker_Tag);
5472         VelMag_e   = config->GetRiemann_Var2(Marker_Tag);
5473         Flow_Dir = config->GetRiemann_FlowDir(Marker_Tag);
5474 
5475         /*--- Non-dim. the inputs if necessary. ---*/
5476         Density_e /= config->GetDensity_Ref();
5477         VelMag_e /= config->GetVelocity_Ref();
5478 
5479         for (auto iDim = 0u; iDim < nDim; iDim++)
5480           Velocity_e[iDim] = VelMag_e*Flow_Dir[iDim];
5481         Energy_e = Energy_i;
5482         break;
5483 
5484       case STATIC_PRESSURE:
5485 
5486         /*--- Retrieve the static pressure for this boundary. ---*/
5487         Pressure_e = config->GetRiemann_Var1(Marker_Tag);
5488         Pressure_e /= config->GetPressure_Ref();
5489         Density_e = Density_i;
5490 
5491         /* --- Compute the boundary state u_e --- */
5492         GetFluidModel()->SetTDState_Prho(Pressure_e, Density_e);
5493         for (auto iDim = 0u; iDim < nDim; iDim++)
5494           Velocity_e[iDim] = Velocity_i[iDim];
5495 
5496         Velocity2_e = GeometryToolbox::SquaredNorm(nDim, Velocity_e);

5502         break;
5503       }

5507

5509       conv_numerics->GetPMatrix_inv(invP_Tensor, &Density_i, Velocity_i, &SoundSpeed_i, &Chi_i, &Kappa_i, UnitNormal);
5510 
5511       /*--- eigenvalues contribution due to grid motion ---*/
5512       su2double ProjVelocity_i = GeometryToolbox::DotProduct(nDim, Velocity_i, UnitNormal);

5515         const su2double ProjGridVel = GeometryToolbox::DotProduct(nDim, gridVel, UnitNormal);
5516         ProjVelocity_i -= ProjGridVel;

5520       su2double Lambda_i[MAXNVAR] = {0}, u_e[MAXNVAR] = {0}, u_i[MAXNVAR]={0}, u_b[MAXNVAR]={0},

5523       u_i[0] = Density_i;
5524       for (auto iDim = 0u; iDim < nDim; iDim++) {

5526         u_e[iDim+1] = Velocity_e[iDim]*Density_e;

5542       for (auto iVar = 0u; iVar < nVar; iVar++) {
5543         u_b[iVar] = u_i[iVar];
5544         for (auto jVar = 0u; jVar < nVar; jVar++) {
5545           if (Lambda_i[jVar] < 0)
5546             u_b[iVar] += P_Tensor[iVar][jVar]*dw[jVar];
5547         }
5548       }
5549 
5550       /*--- Compute the thermodynamic state in u_b ---*/
5551       su2double Velocity_b[MAXNDIM] = {0};
5552       const auto Density_b = u_b[0];
5553       for (auto iDim = 0u; iDim < nDim; iDim++)
5554         Velocity_b[iDim] = u_b[iDim+1]/Density_b;

5559       const bool pressure_boundary = config->GetKind_Data_Riemann(Marker_Tag) == STATIC_PRESSURE;

5563

5567

5570       const su2double Enthalpy_b = Energy_b + Pressure_b/Density_b;
5571       const su2double Kappa_b = GetFluidModel()->GetdPde_rho() / Density_b;
5572       const su2double Chi_b = GetFluidModel()->GetdPdrho_e() - Kappa_b * (StaticEnergy_b+k_b);

5578       /*--- Residual contribution due to grid motion ---*/
5579       if (dynamic_grid) {
5580         const auto gridVel = geometry->nodes->GetGridVel(iPoint);
5581         const auto projVelocity = GeometryToolbox::DotProduct(nDim, gridVel, Normal);
5582         for (auto iVar = 0u; iVar < nVar; iVar++)
5583           Residual[iVar] -= projVelocity *u_b[iVar];
5584       }
5585 
5586       /*--- Update residual value ---*/
5587       LinSysRes.AddBlock(iPoint, Residual);
5588 
5589       if (implicit) {
5590         su2double **Jacobian_b = new su2double*[nVar],
5591                   **Jacobian_i = new su2double*[nVar];
5592         su2double **DubDu = new su2double*[nVar];
5593         for (auto iVar = 0u; iVar < nVar; iVar++){
5594           Jacobian_b[iVar] = new su2double[nVar];
5595           Jacobian_i[iVar] = new su2double[nVar];

5597           DubDu[iVar] = new su2double[nVar];
5598           for (auto jVar=0u; jVar < nVar; jVar++)
5599             DubDu[iVar][jVar] = 0;

5602         }
5603 
5604         /*--- Compute DubDu -= RNL---*/
5605         for (auto iVar=0u; iVar<nVar; iVar++){
5606           for (auto jVar=0u; jVar<nVar; jVar++){

5611           }
5612         }
5613

5623         }
5624

5626         for (auto iVar=0u; iVar<nVar; iVar++) {
5627           for (auto jVar=0u; jVar<nVar; jVar++) {
5628             Jacobian_i[iVar][jVar] = 0.0;
5629             /*--- Compute numerical flux Jacobian at node i ---*/
5630             for (auto kVar=0u; kVar<nVar; kVar++)

5633         }

5639           delete [] Jacobian_b[iVar];
5640           delete [] DubDu[iVar];
5641           delete [] Jacobian_i[iVar];
5642         }
5643         delete [] Jacobian_b;
5644         delete [] DubDu;
5645         delete [] Jacobian_i;
5646       }
5647 
5648       /*--- Viscous contribution ---*/
5649 
5650       if (viscous) {
5651 
5652         /*--- Primitive variables, using the derived quantities ---*/
5653         auto V_boundary= GetCharacPrimVar(val_marker, iVertex);
5654         V_boundary[prim_idx.Temperature()] = Temperature_b;
5655         for (auto iDim = 0u; iDim < nDim; iDim++)
5656           V_boundary[prim_idx.Velocity() + iDim] = Velocity_b[iDim];
5657         V_boundary[prim_idx.Pressure()] = Pressure_b;
5658         V_boundary[prim_idx.Density()] = Density_b;
5659         V_boundary[prim_idx.Enthalpy()] = Enthalpy_b;
5660 
5661         /*--- Set laminar and eddy viscosity at the infinity ---*/
5662 
5663         V_boundary[prim_idx.LaminarViscosity()] = GetFluidModel()->GetLaminarViscosity();
5664         V_boundary[prim_idx.EddyViscosity()] = nodes->GetEddyViscosity(iPoint);
5665         V_boundary[prim_idx.ThermalConductivity()] = GetFluidModel()->GetThermalConductivity();
5666         V_boundary[prim_idx.CpTotal()] = GetFluidModel()->GetCp();
5667 
5668         /*--- Set the normal vector and the coordinates ---*/
5669 
5670         visc_numerics->SetNormal(Normal);
5671         su2double Coord_Reflected[MAXNDIM];
5672         GeometryToolbox::PointPointReflect(nDim, geometry->nodes->GetCoord(Point_Normal),
5673                                                  geometry->nodes->GetCoord(iPoint), Coord_Reflected);
5674         visc_numerics->SetCoord(geometry->nodes->GetCoord(iPoint), Coord_Reflected);
5675 
5676         /*--- Primitive variables, and gradient ---*/
5677 
5678         visc_numerics->SetPrimitive(V_domain, V_boundary);
5679         visc_numerics->SetPrimVarGradient(nodes->GetGradient_Primitive(iPoint), nodes->GetGradient_Primitive(iPoint));
5680 
5681         /*--- Secondary variables ---*/
5682 
5683         const su2double* S_domain = ideal_gas? nullptr : nodes->GetSecondary(iPoint);
5684 
5685         /*--- Compute secondary thermodynamic properties (partial derivatives...) ---*/
5686         su2double S_boundary[8];
5687         S_boundary[0]= GetFluidModel()->GetdPdrho_e();
5688         S_boundary[1]= GetFluidModel()->GetdPde_rho();
5689 
5690         S_boundary[2]= GetFluidModel()->GetdTdrho_e();
5691         S_boundary[3]= GetFluidModel()->GetdTde_rho();
5692 
5693         /*--- Compute secondary thermo-physical properties (partial derivatives...) ---*/
5694 
5695         S_boundary[4]= GetFluidModel()->Getdmudrho_T();
5696         S_boundary[5]= GetFluidModel()->GetdmudT_rho();
5697 
5698         S_boundary[6]= GetFluidModel()->Getdktdrho_T();
5699         S_boundary[7]= GetFluidModel()->GetdktdT_rho();
5700 
5701         visc_numerics->SetSecondary(S_domain, S_boundary);
5702 
5703         /*--- Turbulent kinetic energy ---*/
5704 
5705         if (config->GetKind_Turb_Model() == TURB_MODEL::SST)
5706           visc_numerics->SetTurbKineticEnergy(solver_container[TURB_SOL]->GetNodes()->GetSolution(iPoint,0),
5707                                               solver_container[TURB_SOL]->GetNodes()->GetSolution(iPoint,0));
5708 
5709         /*--- Compute and update residual ---*/
5710         auto residual = visc_numerics->ComputeResidual(config);
5711         LinSysRes.SubtractBlock(iPoint, residual);
5712 
5713         /*--- Jacobian contribution for implicit integration ---*/
5714         if (implicit)
5715           Jacobian.SubtractBlock2Diag(iPoint, residual.jacobian_i);
5716 
5717       }
5718       /*--- Store number of Newton iterations at BC ---*/
5719       if(config->GetKind_FluidModel() == DATADRIVEN_FLUID)
5720         nodes->SetNewtonSolverIterations(iPoint, GetFluidModel()->GetnIter_Newton());
5721 
5722     }
5723   }
5724   END_SU2_OMP_FOR
5725 
5726   for(auto iVar=0u; iVar<nVar; iVar++) {
5727     delete [] P_Tensor[iVar];
5728     delete [] invP_Tensor[iVar];
5729   }
5730   delete[] P_Tensor;
5731   delete[] invP_Tensor;
5732 }
