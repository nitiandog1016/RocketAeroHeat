253:     */

262:

267:         celldata.in_turbulent_zone.length = ncells;
268:         celldata.lengths.length = ncells + nghost;
269: 
270:         celldata.areas.length     = ngtl*(ncells + nghost);
271:         celldata.volumes.length   = ngtl*(ncells + nghost);
272:         celldata.positions.length = ngtl*(ncells + nghost);
273: 
274:         celldata.U0.length = (ncells + nghost)*neq*nftl;
275:         if (nftl>1) celldata.U1.length = (ncells + nghost)*neq*nftl;
276:         if (nftl>2) celldata.U2.length = (ncells + nghost)*neq*nftl;
277:         if (nftl>3) celldata.U3.length = (ncells + nghost)*neq*nftl;
278:         if (nftl>4) celldata.U4.length = (ncells + nghost)*neq*nftl;
279:         celldata.dUdt0.length = (ncells + nghost)*neq*nftl;
280:         if (nftl>1) celldata.dUdt1.length = (ncells + nghost)*neq*nftl;
281:         if (nftl>2) celldata.dUdt2.length = (ncells + nghost)*neq*nftl;
282:         if (nftl>3) celldata.dUdt3.length = (ncells + nghost)*neq*nftl;
283:         if (nftl>4) celldata.dUdt4.length = (ncells + nghost)*neq*nftl;
284:         celldata.source_terms.length = (ncells + nghost)*neq;
285: 
286:         celldata.flowstates.reserve(ncells + nghost);
287:         celldata.gradients.reserve(ncells + nghost);
288:         celldata.workspaces.reserve(ncells + nghost);
289:         foreach (n; 0 .. ncells+nghost) celldata.flowstates ~= FlowState(gmodel, nturb);
290:         foreach (n; 0 .. ncells+nghost) celldata.gradients ~= FlowGradients(myConfig);
291:         foreach (n; 0 .. ncells+nghost) celldata.workspaces ~= WLSQGradWorkspace();
292: 
293:         version(newton_krylov) {
294:             celldata.cell_jacobians.length = (ncells+nghost)*neq*neq;
295:             celldata.saved_gradients.reserve(ncells + nghost);
296:             foreach (n; 0 .. ncells+nghost) celldata.saved_gradients ~= FlowGradients(myConfig);
297:             celldata.saved_source_terms.length = (ncells + nghost)*neq;
298:             celldata.doNotPerturb.length = (ncells+nghost);
299:         }
