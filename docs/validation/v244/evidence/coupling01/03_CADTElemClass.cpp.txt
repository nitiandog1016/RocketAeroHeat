    86     for (unsigned long i = 0; i < val_connElem.size(); ++i) val_connElem[i] += offsetRank;

    89     int sizeGlobal = displs.back() + recvCounts.back();

   105     sizeGlobal = displs.back() + recvCounts.back();

   134     for (int i = 1; i < size; ++i) displs[i] = displs[i - 1] + recvCounts[i - 1];

   137     sizeGlobal = displs.back() + recvCounts.back();
