   575 void CMultiGridIntegration::MultiGrid_Cycle(CGeometry ****geometry,
   576                                             CSolver *****solver_container,
   577                                             CNumerics ******numerics_container,
   578                                             CConfig **config_container,
   579                                             unsigned short iMesh,
   580                                             unsigned short RecursiveParam,
   581                                             unsigned short RunTime_EqSystem,
   582                                             unsigned short iZone,
   583                                             unsigned short iInst) {
   584   SU2_ZONE_SCOPED
   585 
   586   CConfig* config = config_container[iZone];
   587 
   588   const unsigned short Solver_Position = config->GetContainerPosition(RunTime_EqSystem);
   589   const bool implicit = (config->GetKind_TimeIntScheme() == EULER_IMPLICIT);
   590 
   591   /*--- Shorter names to refer to fine grid entities. ---*/
   592 
   593   CGeometry* geometry_fine = geometry[iZone][iInst][iMesh];
   594   CSolver** solver_container_fine = solver_container[iZone][iInst][iMesh];
   595   CSolver* solver_fine = solver_container_fine[Solver_Position];
   596   CNumerics** numerics_fine = numerics_container[iZone][iInst][iMesh][Solver_Position];
   597 
   598   /*--- Number of RK steps. ---*/
   599 
   600   unsigned short iRKLimit = config->GetnRKStep();
   601 
   602   /*--- Do a presmoothing on the grid iMesh to be restricted to the grid iMesh+1 ---*/
   603 
   604   PreSmoothing(RunTime_EqSystem, geometry, solver_container, config_container, solver_fine, numerics_fine,
   605                geometry_fine, solver_container_fine, config, iMesh, iZone, iRKLimit);
   606 
   607   /*--- Compute Forcing Term $P_(k+1) = I^(k+1)_k(P_k+F_k(u_k))-F_(k+1)(I^(k+1)_k u_k)$ and update solution for multigrid ---*/
   608 
   609   if ( iMesh < config->GetnMGLevels() ) {
   610 
   611     /*--- Shorter names to refer to coarse grid entities. ---*/
   612 
   613     CGeometry* geometry_coarse = geometry[iZone][iInst][iMesh+1];
   614     CSolver** solver_container_coarse = solver_container[iZone][iInst][iMesh+1];
   615     CSolver* solver_coarse = solver_container_coarse[Solver_Position];
   616     CNumerics** numerics_coarse = numerics_container[iZone][iInst][iMesh+1][Solver_Position];
   617 
   618     /*--- Temporarily disable implicit integration, for what follows we do not need the Jacobian. ---*/
   619 
   620     if (implicit) {
   621       SU2_OMP_SAFE_GLOBAL_ACCESS(config->SetKind_TimeIntScheme(EULER_EXPLICIT);)
   622     }
   623 
   624     /*--- Compute $r_k = P_k + F_k(u_k)$ ---*/
   625 
   626     solver_fine->Preprocessing(geometry_fine, solver_container_fine, config, iMesh, NO_RK_ITER, RunTime_EqSystem, false);
   627 
   628     Space_Integration(geometry_fine, solver_container_fine, numerics_fine, config, iMesh, NO_RK_ITER, RunTime_EqSystem);
   629 
   630     /*--- LinSysRes = R(u_N) here, before tau is added by SetResidual_Term. ---*/
   631     BEGIN_SU2_OMP_SAFE_GLOBAL_ACCESS {
   632       lastPreSmoothRMS[iMesh][1] = ComputeLinSysResRMS(solver_fine);
   633     }
   634     END_SU2_OMP_SAFE_GLOBAL_ACCESS
   635 
   636     SetResidual_Term(geometry_fine, solver_fine);
   637 
   638     /*--- Compute $r_(k+1) = F_(k+1)(I^(k+1)_k u_k)$ ---*/
   639 
   640     SetRestricted_Solution(RunTime_EqSystem, solver_fine, solver_coarse, geometry_fine, geometry_coarse, config);
   641 
   642     solver_coarse->Preprocessing(geometry_coarse, solver_container_coarse, config, iMesh+1, NO_RK_ITER, RunTime_EqSystem, false);
   643 
   644     Space_Integration(geometry_coarse, solver_container_coarse, numerics_coarse, config, iMesh+1, NO_RK_ITER, RunTime_EqSystem);
   645 
   646     /*--- Compute $P_(k+1) = I^(k+1)_k(r_k) - r_(k+1) ---*/
   647 
   648     SetForcing_Term(solver_fine, solver_coarse, geometry_fine, geometry_coarse, config, iMesh+1);
   649 
   650     /*--- Restore the time integration settings. ---*/
   651 
   652     if (implicit) {
   653       SU2_OMP_SAFE_GLOBAL_ACCESS(config->SetKind_TimeIntScheme(EULER_IMPLICIT);)
   654     }
   655 
   656     /*--- Recursive call to MultiGrid_Cycle (this routine). ---*/
   657     /*--- Execute multigrid cycles sequentially to ensure deterministic recursion order ---*/
   658     /*--- This prevents accumulation of floating-point variations across recursive calls ---*/
   659 
   660     for (unsigned short imu = 0; imu <= RecursiveParam; imu++) {
   661 
   662       unsigned short nextRecurseParam = RecursiveParam;
   663       if (iMesh == config->GetnMGLevels()-2)
   664         nextRecurseParam = 0;
   665 
   666       MultiGrid_Cycle(geometry, solver_container, numerics_container, config_container,
   667                       iMesh+1, nextRecurseParam, RunTime_EqSystem, iZone, iInst);
   668     }
   669 
   670     /*--- Compute prolongated solution, and smooth the correction $u^(new)_k = u_k +
   671           Smooth(I^k_(k+1)(u_(k+1)-I^(k+1)_k u_k))$ ---*/
   672 
   673     GetProlongated_Correction(RunTime_EqSystem, solver_fine, solver_coarse, geometry_fine, geometry_coarse, config);
   674 
   675     const auto& mgOpts = config->GetMGOptions();
   676     SmoothProlongated_Correction(RunTime_EqSystem, solver_fine, geometry_fine, mgOpts.MG_CorrecSmooth[iMesh], mgOpts.MG_Smooth_Coeff, config, iMesh);
   677 
   678     SetProlongated_Correction(solver_fine, geometry_fine, config, iMesh);
   679 
   680     /*--- Solution post-smoothing in the prolongated grid. ---*/
   681 
   682     PostSmoothing(RunTime_EqSystem, solver_fine, numerics_fine, geometry_fine, solver_container_fine,
   683                   config, iMesh, iRKLimit);
   684   }
   685 
   686 }

   689                                              passivedouble defect, const CMGOptions& mgOpts,
   690                                              passivedouble stag_tol, bool early_exit,
   691                                              passivedouble lastRMS[2], char& exitReason,
   692                                              passivedouble& worstStepRatio, unsigned short& worstStep) {
