  1483     case MPI_QUANTITIES::PRIMITIVE_GRADIENT:
  1484     case MPI_QUANTITIES::PRIMITIVE_GRAD_REC:

  1486       MPI_TYPE         = COMM_TYPE::DOUBLE;

  1489       COUNT_PER_POINT  = nPrimVarGrad;
  1490       MPI_TYPE         = COMM_TYPE::DOUBLE;
  1491       break;
  1492     case MPI_QUANTITIES::SOLUTION_EDDY:
  1493       COUNT_PER_POINT  = nVar+1;
  1494       MPI_TYPE         = COMM_TYPE::DOUBLE;
  1495       break;
  1496     case MPI_QUANTITIES::STOCH_SOURCE_LANG:
  1497       COUNT_PER_POINT  = nDim;
  1498       MPI_TYPE         = COMM_TYPE::DOUBLE;
  1499       break;
  1500     case MPI_QUANTITIES::DES_LENGTHSCALE:
  1501       COUNT_PER_POINT  = 1;
  1502       MPI_TYPE         = COMM_TYPE::DOUBLE;
  1503       break;
  1504     case MPI_QUANTITIES::SOLUTION_FEA:
  1505       if (config->GetTime_Domain())
  1506         COUNT_PER_POINT  = nVar*3;
  1507       else
  1508         COUNT_PER_POINT  = nVar;
