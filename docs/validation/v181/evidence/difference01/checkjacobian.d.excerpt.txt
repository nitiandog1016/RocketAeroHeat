145:      * 0. Initialize Newton-Krylov simulation
146:      */
147:     initNewtonKrylovSimulation(snapshot, 1, 1, "1000000");
148:     JSONValue jsonData = readJSONfile(lmrCfg.nkCfgFile);
149:     nkCfg.readValuesFromJSON(jsonData);
150:     // Allocate space and configure phases
151:     nkPhases.length = nkCfg.numberOfPhases;
152:     foreach (i, ref phase; nkPhases) {
153:         string key = "NewtonKrylovPhase_" ~ to!string(i);
154:         phase.readValuesFromJSON(jsonData[key]);
155:     }
156:     activePhase = nkPhases[$-1]; // set to the last phase from the simulation
157:     if (readFrozenShockDetectorValues) readShockDetectorValues(); // we have to set the frozen shock detector flags before the residual evaluation below
158:     evalResidualWorker(0); // we perform a residual evaluation here to populate the ghost-cells
159:     alias cfg = GlobalConfig;
160:     size_t nConserved = cfg.cqi.n;
161: 
162:     if (verbosity > 1) writefln("%s: Performing check with Jacobian perturbation parameter %.4e and Frechet perturbation parameter %.4e",
163:                                 cmdName, nkCfg.preconditionerPerturbation, nkCfg.frechetDerivativePerturbation);
164: 
165:     // prepare fluidblock (note that this program only operates on a single fluidblock)
166:     auto blk = localFluidBlocks[0];
167:     if (readFrozenLimiterValues) readLimiterValues(snapshot);
168: 
169:     /*
170:      * 1. Do calculation of sparse-matrix Jacobian by test vector.
171:      */
172:     // 1a. Populate Jacobian
173:     blk.initialize_jacobian(cfg.interpolation_order, nkCfg.preconditionerPerturbation, 0);
174:     blk.evaluate_jacobian(nkCfg.fluxCalculatorForPreconditioner);
175:     // 1b. Prepare a test vector
176:     double[] testVec;
177:     testVec.length = blk.flowJacobian.local.ia.length-1;
178:     Mt19937 rng;
179:     double mag = 0.0;
180:     foreach (ref val; testVec) {
181:         val = rng.uniform01();
182:         mag += val*val;
183:     }
184:     testVec[] = (1./mag)*testVec[]; // we have observed that a normalized test vector behaves better
185: 
186:     // 1c. Compute c0 = J*testVec
187:     double[] c0;
188:     c0.length = testVec.length;
189:     multiply(blk.flowJacobian.local, testVec, c0);
190: 
191:     /*
192:      * 2. Do calculation of Jv with Frechet derivative.
193:      */
194:     // 2a. Populate z with test vector
195:     allocateGlobalGMRESWorkspace();
196:     blk.allocate_GMRES_workspace(1, nkCfg.useRealValuedFrechetDerivative);
197: 
198:     // 2b. Do Frechet step (note for the real-valued variant we need to fill R0)
199:     version(complex_numbers) {
200:         // do nothing
201:     } else {
202:         int startIdx = 0;
203:         foreach (cell; blk.cells) {
204:             foreach (ivar; 0 .. nConserved) { blk.R0[startIdx+ivar] = cell.dUdt[0][ivar].re; }
205:             startIdx += nConserved;
206:         }
207:     }
208:     blk.zed[] = testVec[];
209:     evalJacobianVectorProduct(nkCfg.frechetDerivativePerturbation, 0, 0);
210: 
211:     /*
212:      * 3. Compare vectors and write result to file.
213:      */
214:     double c0_2norm = 0.0;
215:     foreach (val; c0) c0_2norm += val*val;
216:     c0_2norm = sqrt(c0_2norm);
217: 
218:     double c1_2norm = 0.0;
219:     foreach (val; blk.zed) c1_2norm += val*val;
220:     c1_2norm = sqrt(c1_2norm);
221: 
222:     writefln("c0, 2-norm= %.16e", c0_2norm.re);
223:     writefln("c1, 2-norm= %.16e", c1_2norm.re);
224: 
225:     outfile.writeln("# array_idx cell_idx c0   c1   c0-c1   abs(c0-c1)");
226:     foreach (i; 0 .. c0.length) {
227:         auto delta = c0[i] - blk.zed[i];
228:         size_t cell_id = i/nConserved;
229:         outfile.writefln("%d %d %20.16e %20.16e %20.16e %20.16e", i, cell_id, c0[i].re, blk.zed[i].re, delta.re, fabs(delta).re);
230:     }
231:     outfile.close();
232: 
233:     return 0;
234: }
