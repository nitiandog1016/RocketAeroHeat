def read_json(path):
    return json.loads(path.read_bytes())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def official_nodes():
    archive = locked(DATA/'raw/Axiswblim7_grids.zip','6d12b620b0e1fff13c0567a9ad4f18c88133ed9d87ee332be8c9d775c7cdc4aa')
    with zipfile.ZipFile(io.BytesIO(archive)) as z:
        members = [name for name in z.namelist() if name.endswith('/kussoy_2zn_2levdn.p2dfmt.gz')]
        assert len(members) == 1
        compressed = z.read(members[0])
    assert hashlib.sha256(compressed).hexdigest() == '8980c3ba3c2ebdfdd661617dc27380df2af9a3ed45822fa8b4a1dba1596f11d8'
    data = np.fromstring(gzip.decompress(compressed).decode(), sep=' ')
    assert np.array_equal(data[:5],[2,81,101,81,101])
    blocks = data[5:].reshape(4,101,81).transpose(0,2,1)
    assert np.array_equal(blocks[0:2,-1],blocks[2:,0])
    return -0.01*np.vstack((blocks[0],blocks[2,1:])), 0.01*np.vstack((blocks[1],blocks[3,1:]))


def norm_rate(mesh, info, scale):
    # 由保存的面拥有者/邻居独立累加五个分量，重算原空间残差。
    faces = mesh['faces']
    flux = info['faceFluxIntegralPerRadian']
    neighbour = faces['neighbour'].astype(int)
    outward = np.zeros_like(info['cellSourceIntegral'])
    for i, owner in enumerate(faces['owner'].astype(int)):
        outward[owner-1] += flux[i]
        if neighbour[i]>0:
            outward[neighbour[i]-1] -= flux[i]
    rate = (info['cellSourceIntegral']-outward) / mesh['volumePerRadian'][:,None]
    return np.max(np.abs(rate)/scale,axis=0)


def verify_chain(folder, saved):
    start = loadmat(folder/'initial.mat',simplify_cells=True)
    mesh, result = saved['mesh'], saved['implicit']
    x, r = official_nodes()
    i, j = start['iNodes'].astype(int)-1, start['jNodes'].astype(int)-1
    for name, official in [('X',x),('R',r)]:
        assert np.array_equal(mesh[name], start['mesh'][name])
        assert np.array_equal(mesh[name],official[np.ix_(i,j)])
    assert np.array_equal(j,np.arange(0,101,4)) and np.all(np.isin(np.arange(0,161,4),i))
    parent_path = ROOT/'outputs/v66_validation/tmr_wall_thermal/tmr_wall_thermal_evidence.mat'
    assert sha(parent_path) == 'd1b8d4479b23358611c653fd131e96759cf9556ed6555fbaea8c8a517694016f'
    parent = loadmat(parent_path,simplify_cells=True,variable_names=['implicit'])['implicit']
    parent_ids = (i[:-1]//4)[:,None]+40*np.arange(25)[None,:]
    assert np.array_equal(start['parentIDs'],parent_ids.flatten(order='F')+1)
    assert np.array_equal(start['initialQ'],parent['Q'][parent_ids.flatten(order='F')])
    volume = mesh['volumePerRadian']
    scale = start['scale']
    old = start['initialQ']
    initial = (volume[:,None]*old).sum(axis=0)
    budget = volume.sum()*scale
    max_balance = 0.
    steps = sorted(folder.glob('step_*.mat'))
    assert len(steps) == saved['report']['acceptedSteps'] and len(steps)>0
    all_residuals = [float(np.max(abs(start['rate'])/scale))]
    old_net = old_src = np.zeros(5)
    max_step_balance = 0.
    gas = saved['gas']
    t_min, t_max = float('inf'), -float('inf')
    for n, file in enumerate(steps,1):
        frame = loadmat(file,simplify_cells=True)
        assert file.name == f'step_{n:03d}.mat' and frame['steps']==n
        q = frame['Q']
        assert q.shape == old.shape and np.isfinite(q).all() and np.all(q[:,0]>0) and np.all(q[:,4]>=0)
        t = (q[:,3]/q[:,0] - 0.5*np.sum((q[:,1:3]/q[:,0,None])**2,axis=1))/gas['cv']
        assert np.isfinite(t).all() and gas['temperatureRange'][0]<=t.min()<=t.max()<=gas['temperatureRange'][1]
        t_min, t_max = min(t_min,float(t.min())), max(t_max,float(t.max()))
        integral = (q*volume[:,None]).sum(axis=0)
        net, src = frame['net'], frame['src']
        balance = np.max(np.abs(integral-initial+net-src)/budget)
        step_balance = np.max(np.abs(((q-old)*volume[:,None]).sum(axis=0)+(net-old_net)-(src-old_src))/budget)
        max_balance = max(max_balance,float(balance))
        max_step_balance = max(max_step_balance,float(step_balance))
        assert frame['history']['newtonDefect'][-1] <= frame['history']['newtonTarget'][-1]
        assert np.array_equal(frame['history']['residual'],[all_residuals[-1],frame['residual']])
        old, old_net, old_src = q, net, src
        all_residuals.append(frame['residual'])
    assert max(max_step_balance,max_balance) <= 5e-10 and np.array_equal(old,result['Q'])
    assert np.array_equal(all_residuals,result['history']['residual'])
    components = norm_rate(mesh,result['finalInfo'],scale)
    assert math.isclose(float(max(components)),result['residualNorm'],rel_tol=1e-6,abs_tol=1e-10)
    steady = len(steps)>=3 and max(all_residuals[-3:])<=2e-6
    assert bool(result['steadyConverged']) == steady
    return dict(acceptedEndpoints=len(steps),maximumGlobalBalance=max_balance,
                maximumStepBalance=max_step_balance,recomputedFinalScaledResiduals=components.tolist(),
                lastThreeResiduals=all_residuals[-3:],steadyCriterionMet=bool(steady),
                acceptedCellTemperatureRangeK=[t_min,t_max])


def audit(folder,out):
    receipt = read_json(folder/'debugger_process.json')
    assert receipt['status']=='terminal' and receipt['debuggerExitCode']==0 and receipt['stopReason'] is None
    assert not receipt['survivingProcessIDs'] and all(receipt['inputsStable'].values())
    saved = loadmat(folder/'terminal.mat',simplify_cells=True)
    chain = verify_chain(folder,saved)
    m = saved['mesh']
    w = saved['implicit']['finalInfo']['wall']
    order = np.argsort(w['ownerCell'])
    x, r = m['X'][:,0], m['R'][:,0]
    n = len(x)-1
    assert np.array_equal(w['ownerCell'][order],np.arange(1,n+1)) and np.all(np.diff(x)<0)
    fid = w['faceID'][order].astype(int)
    assert np.all(m['faces']['boundarySide'][fid-1]==3) and np.all(w['temperatureInput']==311)
    area = np.pi*(r[:-1]+r[1:])*np.hypot(np.diff(r),np.diff(x))
    np.testing.assert_allclose(area,2*np.pi*w['areaPerRadian'][order],rtol=2e-12,atol=1e-14)
    pressure, heat = w['pressure'][order], w['heatFluxIntoWall'][order]
    owner = w['ownerCell'][order].astype(int)-1
    cells = saved['implicit']['Q'][owner]
    temperature = (cells[:,3]/cells[:,0] - 0.5*np.sum((cells[:,1:3]/cells[:,0,None])**2,axis=1))/saved['gas']['cv']
    np.testing.assert_allclose(temperature,saved['implicit']['finalInfo']['primitive'][owner,3],rtol=2e-12)
    assert np.isfinite(pressure).all() and np.isfinite(heat).all()
    condition = json.loads(locked(DATA/'contract.json',HASHES['contract']))
    ref = condition['experimental_local_reference']
    obs = {k: measurements(DATA/'raw'/file,k,Decimal(str(ref[name]))) for k,file,name in
           [('p','pw_exp.dat','pressure_Pa'),('q','qw_exp.dat','reported_wall_heating_magnitude_W_m2')]}
    prior = ROOT/'docs/validation/v263/evidence/tmr_joint/wall_stations.csv'
    with prior.open(encoding='utf-8-sig',newline='') as stream:
        base = {Decimal(row['S_cm']):row for row in csv.DictReader(stream) if row['grid']=='baseline'}
    rows=[];errors={'p':[],'q':[]};paired=[];owner={'p':set(),'q':set()}
    for s in sorted(set(obs['p'])|set(obs['q'])):
        project_x = -float(s/100)*(math.cos(math.radians(20)) if s>0 else 1)
        face = int(np.searchsorted(-x,-project_x,side='right'))-1
        assert 0<=face<n and x[face]>=project_x>x[face+1]
        row = dict(S_cm=str(s),wallFaceID=int(fid[face]),pressure_Pa=float(pressure[face]),
                   heatIntoWall_W_m2=float(heat[face]))
        for k,values,meas in [('p',pressure,'expPressure_Pa'),('q',heat,'expHeatingMagnitude_W_m2')]:
            sample = obs[k].get(s)
            relative = (float(values[face])/sample['value']-1) if sample else None
            row[meas] = sample['value'] if sample else None
            row[k+'_source_rows'] = '|'.join(map(str,sample['rows'])) if sample else ''
            row[k+'_relative_error'] = relative
            if sample:
                assert face not in owner[k]
                owner[k].add(face)
                errors[k].append((relative,float(base[s][k+'_relative_error'])))
        row['both_inside_estimates']=None
        if s in obs['p'] and s in obs['q']:
            row['both_inside_estimates']=all(abs(row[k+'_relative_error'])<=.1 for k in ('p','q'))
            paired.append(row)
        rows.append(row)
    assert [len(rows),len(errors['p']),len(errors['q']),len(paired)]==[39,24,27,12]
    out.mkdir(exist_ok=False)
    write_csv(out/'stations.csv',rows)
    rings=[dict(segment=k+1,faceID=int(fid[k]),X1_m=x[k],R1_m=r[k],X2_m=x[k+1],R2_m=r[k+1],
                ringArea_m2=area[k],pressure_Pa=pressure[k],heatIntoWall_W_m2=heat[k],givenTw_K=311,
                solvedSolidTw_K=None,nearWallCellID=int(owner[k])+1,nearWallCellT_K=temperature[k],
                method='axisymmetric SA-noft2; Sutherland; dirichlet-tangent',
                frame='terminal',physicalQualification=False) for k in range(n)]
    write_csv(out/'wall.csv',rings)
    rms={key:dict(zip(('current','baseline'),np.sqrt(np.mean(np.square(values),axis=0)).tolist()))
         for key,values in errors.items()}
    positions=[float(r['S_cm']) for r in paired if r['both_inside_estimates']]
    summary=dict(case=condition['case_id'],meshCells=int(m['cellCount']),wallRings=n,wallAreaM2=math.fsum(area),
                 signedHeatPowerW=float(np.dot(area,heat)),chain=chain,allStationRelativeRMS=rms,
                 jointStations=len(paired),insideAtJointStationsCm=positions,allSameVariableStationsSeparated=True,
                 solver=saved['report'],terminalMAT_SHA256=sha(folder/'terminal.mat'),referenceCSV_SHA256=sha(prior),
                 physicalQualification=False,newRocketAreaM2=0)
    (out/'summary.json').write_text(json.dumps(summary,indent=2,allow_nan=False)+'\n',encoding='utf8')
    print(json.dumps(summary,ensure_ascii=True))


if __name__=='__main__':
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('run_folder',type=Path)
    parser.add_argument('new_output_folder',type=Path)
    args=parser.parse_args()
    audit(args.run_folder,args.new_output_folder)
